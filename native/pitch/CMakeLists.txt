cmake_minimum_required(VERSION 3.22)
project(OxidalPitchLV2 VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)

# The TONE3000 pitch engine as a standalone LV2 effect, built from the same
# pinned TONE3000 and JUCE checkouts as the native adapter (native/tone3000).
# Nothing is vendored or downloaded here. @see docs/native-pitch.md
set(OXIDAL_PITCH_JUCE_SOURCE "" CACHE PATH "Pinned JUCE 9.0.3 checkout")
set(OXIDAL_PITCH_TONE3000_SOURCE "" CACHE PATH "Pinned TONE3000 checkout (see native/tone3000/sources.json)")
set(OXIDAL_PITCH_LV2_INSTALL_DIR "lib/lv2" CACHE PATH "LV2 bundle directory (relative to CMAKE_INSTALL_PREFIX unless absolute)")
if(NOT EXISTS "${OXIDAL_PITCH_JUCE_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set OXIDAL_PITCH_JUCE_SOURCE to the pinned JUCE checkout")
endif()
file(READ "${OXIDAL_PITCH_JUCE_SOURCE}/CMakeLists.txt" juce_project)
if(NOT juce_project MATCHES "project\\(JUCE VERSION 9\\.0\\.3 ")
    message(FATAL_ERROR "Oxidal Pitch requires the pinned JUCE 9.0.3")
endif()
set(engine_header "${OXIDAL_PITCH_TONE3000_SOURCE}/plugin/include/PitchShift.h")
set(engine_source "${OXIDAL_PITCH_TONE3000_SOURCE}/plugin/src/PitchShift.cpp")
function(check_pitch_hash path expected)
    file(SHA256 "${path}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Pinned pitch source hash differs: ${path}")
    endif()
endfunction()
# TONE3000 ab5e5ae (v0.0.12): the engine is used unmodified.
check_pitch_hash("${engine_header}" 6dfbd00a6c74b0020ba72e6a5f58c0b98d2c98f4c7975a7fcb9d1be6c36d0c30)
check_pitch_hash("${engine_source}" 069b6c62b96063c04d8fa574c7c8e5e249801acc65fb5e600eefe96194f63e3a)
set(JUCE_MODULES_ONLY ON CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${OXIDAL_PITCH_JUCE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/juce" EXCLUDE_FROM_ALL)
set(lv2_sdk "${OXIDAL_PITCH_JUCE_SOURCE}/modules/juce_audio_processors_headless/format_types/LV2_SDK/lv2")
if(NOT EXISTS "${lv2_sdk}/lv2/core/lv2.h")
    message(FATAL_ERROR "JUCE's pinned LV2 SDK headers are missing")
endif()
function(configure_pitch_target target)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${lv2_sdk}" "${OXIDAL_PITCH_TONE3000_SOURCE}/plugin/include")
    target_compile_definitions(${target} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
        JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0
        JUCE_USE_MP3AUDIOFORMAT=0 JUCE_USE_LAME_AUDIO_FORMAT=0 JUCE_USE_OPUS=0)
    target_link_libraries(${target} PRIVATE juce::juce_dsp)
    set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -fno-fast-math -fvisibility=hidden)
    endif()
endfunction()
add_library(oxidal_pitch MODULE OxidalPitch.cpp "${engine_source}")
configure_pitch_target(oxidal_pitch)
set(bundle "${CMAKE_CURRENT_BINARY_DIR}/oxidal-pitch.lv2")
set_target_properties(oxidal_pitch PROPERTIES PREFIX "" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${bundle}")
file(MAKE_DIRECTORY "${bundle}")
foreach(file manifest.ttl pitch.ttl NOTICE)
    configure_file("${file}" "${bundle}/${file}" COPYONLY)
endforeach()
configure_file("${OXIDAL_PITCH_TONE3000_SOURCE}/LICENSE" "${bundle}/TONE3000-LICENSE" COPYONLY)
configure_file("${OXIDAL_PITCH_JUCE_SOURCE}/LICENSE.md" "${bundle}/JUCE-LICENSE.md" COPYONLY)
install(TARGETS oxidal_pitch LIBRARY DESTINATION "${OXIDAL_PITCH_LV2_INSTALL_DIR}/oxidal-pitch.lv2")
install(FILES manifest.ttl pitch.ttl NOTICE "${bundle}/TONE3000-LICENSE" "${bundle}/JUCE-LICENSE.md"
    DESTINATION "${OXIDAL_PITCH_LV2_INSTALL_DIR}/oxidal-pitch.lv2")

if(BUILD_TESTING)
    # The wrapper is checked sample for sample against the engine compiled on its own.
    add_executable(pitch_wrapper_test tests/PitchWrapperTest.cpp tests/UpstreamReference.cpp OxidalPitch.cpp "${engine_source}")
    configure_pitch_target(pitch_wrapper_test)
    target_compile_definitions(pitch_wrapper_test PRIVATE
        PITCH_REFERENCE_CPP="${engine_source}"
        PITCH_REFERENCE_HEADER="${engine_header}")
    add_test(NAME pitch-wrapper-upstream COMMAND pitch_wrapper_test)
    add_executable(pitch_bundle_test tests/LoadBundleTest.cpp)
    target_compile_features(pitch_bundle_test PRIVATE cxx_std_20)
    target_include_directories(pitch_bundle_test PRIVATE "${lv2_sdk}")
    target_link_libraries(pitch_bundle_test PRIVATE ${CMAKE_DL_LIBS})
    add_dependencies(pitch_bundle_test oxidal_pitch)
    add_test(NAME pitch-bundle-load COMMAND pitch_bundle_test "$<TARGET_FILE:oxidal_pitch>")
endif()
