cmake_minimum_required(VERSION 3.22)
project(OxidalSwellLV2 VERSION 0.1.0 LANGUAGES CXX)
include(CTest)

# Oxidal Swell: an original attack-triggered volume swell. Needs only the LV2 headers.
# @see docs/native-swell.md
set(OXIDAL_SWELL_LV2_INSTALL_DIR "lib/lv2" CACHE PATH "LV2 bundle directory (relative to CMAKE_INSTALL_PREFIX unless absolute)")
find_path(LV2_INCLUDE_DIR lv2/core/lv2.h REQUIRED)
function(configure_swell_target target)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${LV2_INCLUDE_DIR}")
    set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fno-fast-math -fvisibility=hidden)
    endif()
endfunction()
add_library(oxidal_swell MODULE OxidalSwell.cpp)
configure_swell_target(oxidal_swell)
set(bundle "${CMAKE_CURRENT_BINARY_DIR}/oxidal-swell.lv2")
set_target_properties(oxidal_swell PROPERTIES PREFIX "" SUFFIX ".so" LIBRARY_OUTPUT_DIRECTORY "${bundle}")
file(MAKE_DIRECTORY "${bundle}")
foreach(file manifest.ttl swell.ttl NOTICE)
    configure_file("${file}" "${bundle}/${file}" COPYONLY)
endforeach()
install(TARGETS oxidal_swell LIBRARY DESTINATION "${OXIDAL_SWELL_LV2_INSTALL_DIR}/oxidal-swell.lv2")
install(FILES manifest.ttl swell.ttl NOTICE DESTINATION "${OXIDAL_SWELL_LV2_INSTALL_DIR}/oxidal-swell.lv2")

if(BUILD_TESTING)
    add_executable(swell_test tests/SwellTest.cpp OxidalSwell.cpp)
    configure_swell_target(swell_test)
    add_test(NAME swell-detect-curve-time-stereo COMMAND swell_test)
endif()
